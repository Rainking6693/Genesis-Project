#!/usr/bin/env python3
"""
DreamGym safe trainer (finite-cycle aware)

Runs repeated synthetic training cycles using the in-repo DreamGymTrainer
and saves a checkpoint to checkpoints/latest_god_mode.json.

Env overrides:
  MAX_CYCLES   - number of cycles to run (default: 50)
  BATCH_SIZE   - synthetic batch size per task (default: 64)
  TASKS        - comma-separated task signatures (default: refactor,seo,deploy,marketing)
  SLEEP_SEC    - sleep between cycles (default: 100)
"""

import json
import os
import time
import random
from pathlib import Path

from infrastructure.dreamgym.integration import DreamGymTrainer
from infrastructure.dreamgym.experience_model import DreamGymExperience


def _make_real_experience(task: str, difficulty: str, reward: float, novelty: float) -> DreamGymExperience:
    """Create a pseudo-real experience to mix into the buffer with variable reward/novelty."""
    ts = time.time()
    return DreamGymExperience(
        task_signature=task,
        difficulty=difficulty,
        synthetic=False,
        observation=f"[REAL {difficulty}] {task} observation at {ts}",
        action=f"real_action::{task}::{difficulty}",
        reward=round(max(0.0, min(0.99, reward)), 3),
        novelty_score=round(max(0.0, min(0.99, novelty)), 3),
        generated_at=time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime(ts)),
        metadata={"source": "pseudo_real", "timestamp": ts},
    )


def run_cycle(trainer: DreamGymTrainer, tasks, batch_size: int, difficulty: str, cycle_idx: int, max_cycles: int):
    # 1) synthetic batch per task
    for task in tasks:
        trainer.generate_synthetic_batch(task, batch_size=batch_size)
        # 2) inject a few pseudo-real trajectories to mix ratios
        for _ in range(max(2, batch_size // 16)):
            # reward/novelty curriculum: ramp up over cycles with noise
            progress = (cycle_idx) / max(1, max_cycles - 1)
            base_reward = 0.55 + 0.35 * progress  # 0.55 -> 0.9
            base_novelty = 0.70 + 0.20 * progress  # 0.70 -> 0.90
            reward_noise = random.uniform(-0.05, 0.05)
            novelty_noise = random.uniform(-0.04, 0.04)
            trainer.buffer.add(
                _make_real_experience(
                    task,
                    difficulty,
                    reward=base_reward + reward_noise,
                    novelty=base_novelty + novelty_noise,
                ),
                source="real",
            )

    # 3) prepare evolution batch with 50/50 mix
    prep = trainer.prepare_evolution_batch(tasks[0], batch_size=batch_size, synthetic_ratio=0.5)
    stats = trainer.stats()

    # reward / novelty aggregates
    rewards, novelties = [], []
    for e in prep:
        if isinstance(e, dict):
            if "reward" in e:
                rewards.append(e["reward"])
            if "novelty_score" in e:
                novelties.append(e["novelty_score"])
        else:
            if hasattr(e, "reward"):
                rewards.append(e.reward)
            if hasattr(e, "novelty_score"):
                novelties.append(e.novelty_score)
    reward_avg = sum(rewards) / len(rewards) if rewards else 0.0
    novelty_avg = sum(novelties) / len(novelties) if novelties else 0.0

    return {
        "stats": stats,
        "sample": prep[:3],
        "reward_avg": reward_avg,
        "novelty_avg": novelty_avg,
    }


def save_brain(brain: dict, path: Path):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(brain, indent=2))


def main():
    max_cycles = int(os.getenv("MAX_CYCLES", "1"))
    batch_size = int(os.getenv("BATCH_SIZE", "64"))
    tasks = os.getenv("TASKS", "refactor,seo,deploy,marketing").split(",")
    sleep_sec = int(os.getenv("SLEEP_SEC", "300"))

    # Difficulty schedule cycles through novice -> intermediate -> expert
    difficulties = ["novice", "intermediate", "expert"]

    checkpoint = Path("checkpoints/latest_god_mode.json")

    print(f"DreamGym trainer starting with tasks={tasks}, batch_size={batch_size}, cycles={max_cycles}")

    prev_reward = prev_novelty = None

    for cycle in range(max_cycles):
        trainer = DreamGymTrainer(agent_name="forever-trainer")
        difficulty = difficulties[cycle % len(difficulties)]
        brain = run_cycle(trainer, tasks, batch_size, difficulty, cycle, max_cycles)
        brain["timestamp"] = time.strftime("%Y-%m-%d %H:%M:%S")
        brain["cycle"] = cycle + 1
        brain["difficulty"] = difficulty

        # deltas
        if prev_reward is not None:
            brain["reward_delta"] = brain["reward_avg"] - prev_reward
            brain["novelty_delta"] = brain["novelty_avg"] - prev_novelty
        else:
            brain["reward_delta"] = 0.0
            brain["novelty_delta"] = 0.0

        prev_reward, prev_novelty = brain["reward_avg"], brain["novelty_avg"]

        save_brain(brain, checkpoint)
        print(
            f"[cycle {cycle+1}] diff={difficulty} "
            f"reward_avg={brain['reward_avg']:.3f} (Δ {brain['reward_delta']:+.3f}) "
            f"novelty_avg={brain['novelty_avg']:.3f} (Δ {brain['novelty_delta']:+.3f})"
        )
        if cycle + 1 < max_cycles:
            time.sleep(sleep_sec)

    print("Training run complete.")


if __name__ == "__main__":
    main()
